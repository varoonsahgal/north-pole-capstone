# ---- Firewall for the containers: ONLY the load balancer may talk to them ----
resource "aws_security_group" "tasks" {
  name        = "${var.team}-tasks-sg"
  description = "Allow port 3000 from the load balancer only"
  vpc_id      = data.aws_vpc.default.id

  ingress {
    from_port       = 3000
    to_port         = 3000
    protocol        = "tcp"
    security_groups = [aws_security_group.alb.id] # a security group as the source, not an IP range
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

# The service keeps 3 copies of our task running at all times
resource "aws_ecs_service" "app" {
  name            = "${var.team}-service"
  cluster         = aws_ecs_cluster.main.id
  task_definition = aws_ecs_task_definition.app.arn
  launch_type     = "FARGATE"
  desired_count   = 3

  network_configuration {
    subnets          = data.aws_subnets.default.ids
    assign_public_ip = true                           # needed so the task can pull its image from ECR
    security_groups  = [aws_security_group.tasks.id] # NEW: use our own firewall
  }

  # Register every task in the load balancer's target group
  load_balancer {
    target_group_arn = aws_lb_target_group.app.arn
    container_name   = "sleigh-list"
    container_port   = 3000
  }

  depends_on = [aws_lb_listener.http]
}
